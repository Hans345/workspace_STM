56void fft(Complex c[], int N) {
57 if (N <= 1)
58 return;
59
60 // Aufteilen
61 Complex ce[N / 2];
62 Complex co[N / 2];
63 for (int i = 0; i < N / 2; i++) {
64 ce[i] = c[2 * i];
65 co[i] = c[2 * i + 1];
66 }
67
68 // Rekursion
69 fft(ce, N / 2);
70 fft(co, N / 2);
71
72 // Kombinieren
73 for (int k = 0; k < N / 2; k++) {
74 float angle = (-2.0 * M_PI * k) / N;
75 Complex twiddle = complex_exp(angle);
76
77 Complex t = complex_mul(twiddle, co[k]);
78
79 c[k] = complex_add(ce[k], t);
80 c[k + N / 2] = complex_sub(ce[k], t);
81 }
82}
